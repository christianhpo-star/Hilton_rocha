function renderRanking(){ const root=qs('rankingContainer');if(!root)return; const rooms={};roomCodes().forEach(c=>rooms[c]={uniforme:40,sapato:30,entrada:20,sala:10,bonus:0,falsePenalty:0}); const monthOk=t=>activeRankingMonth==='ALL'||String(t||'').split('/')[1]?.trim()===activeRankingMonth; // historyList contém somente ocorrências aprovadas pela EEB/Gestão. (appData.historyList||[]).forEach(i=>{ if(!rooms[i.roomCode]||!monthOk(i.timestamp))return; const r=String(i.rule||'').toLowerCase(); if(r.includes('uniforme'))rooms[i.roomCode].uniforme=Math.max(0,rooms[i.roomCode].uniforme-1); else if(r.includes('sapato'))rooms[i.roomCode].sapato=Math.max(0,rooms[i.roomCode].sapato-1); else if(r.includes('atraso')||r.includes('entrada'))rooms[i.roomCode].entrada=Math.max(0,rooms[i.roomCode].entrada-1); else if(r.includes('sala'))rooms[i.roomCode].sala=Math.max(0,rooms[i.roomCode].sala-1); }); (appData.bonificacoesList||[]).forEach(b=>{if(rooms[b.roomCode]&&monthOk(b.timestamp))rooms[b.roomCode].bonus+=Number(b.points)||0}); (appData.falseReportsList||[]).forEach(f=>{ const referenceDate=f.originalTimestamp||f.timestamp; if(f.reporterRoom&&rooms[f.reporterRoom]&&monthOk(referenceDate))rooms[f.reporterRoom].falsePenalty+=Number(f.penaltyPoints)||0; }); const total=s=>s.uniforme+s.sapato+s.entrada+s.sala+s.bonus-s.falsePenalty; const sorted=Object.entries(rooms).sort((a,b)=>total(b[1])-total(a[1])); root.innerHTML=''; sorted.forEach(([code,s],idx)=>{ const score=total(s),card=document.createElement('article');card.className=`ranking-card ${idx===0?'leader':''}`; const extras=[]; if(s.bonus)extras.push(`+${s.bonus} bônus`); if(s.falsePenalty)extras.push(`-${s.falsePenalty} denúncia falsa`); card.innerHTML=`
${idx===0?'🏆':idx+1+'º'}
${esc(code)}${esc((appData.roomMapping||{})[code]||code)}
${score} pts
Uniforme ${s.uniforme}/40 Sapato ${s.sapato}/30 Entrada ${s.entrada}/20 Sala ${s.sala}/10${extras.join('')}
`; root.appendChild(card); }); }
