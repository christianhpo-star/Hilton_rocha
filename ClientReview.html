function setReviewFilter(filter,btn){ activeReviewFilter=filter; document.querySelectorAll('.review-filter').forEach(b=>b.classList.remove('active')); if(btn)btn.classList.add('active'); renderReviewQueue(); } function replaceReportLocally(updated){ if(!updated)return; appData.reviewQueueList=(appData.reviewQueueList||[]).map(r=>String(r.id)===String(updated.id)?updated:r); appData.myReportsList=(appData.myReportsList||[]).map(r=>String(r.id)===String(updated.id)?updated:r); } function refreshPendingCounters(){ appData.pendingReviewCount=(appData.reviewQueueList||[]).filter(r=>String(r.status||'').toUpperCase()==='PENDENTE').length; const count=Number(appData.pendingReviewCount)||0; if(qs('pendingReviewCount'))qs('pendingReviewCount').textContent=count; if(qs('navPendingBadge')){qs('navPendingBadge').textContent=count;qs('navPendingBadge').style.display=appData.isEEB&&count>0?'inline-flex':'none'} } function renderReviewQueue(){ const root=qs('reviewQueueContainer');if(!root)return; let reports=[...(appData.reviewQueueList||[])]; if(activeReviewFilter!=='TODAS')reports=reports.filter(r=>String(r.status||'').toUpperCase()===activeReviewFilter); reports.sort((a,b)=>String(a.status)==='PENDENTE'&&String(b.status)!=='PENDENTE'?-1:String(b.status)==='PENDENTE'&&String(a.status)!=='PENDENTE'?1:0); if(!reports.length){ root.innerHTML=`
${activeReviewFilter==='PENDENTE'?'Nenhuma denúncia aguardando análise.':'Nenhum registro neste filtro.'}
`; refreshPendingCounters();return; } root.innerHTML=reports.map(r=>{ const status=String(r.status||'PENDENTE').toUpperCase(); const pending=status==='PENDENTE'; const obs=r.obs?`
${esc(r.obs)}
`:''; const decision=!pending&&r.reviewReason?`
Decisão: ${esc(r.reviewReason)}${r.reviewerEmail?`
Revisado por ${esc(r.reviewerEmail)} · ${esc(r.reviewedAt||'')}`:''}
`:''; const actions=pending?`
`:`
${esc(status)}
`; return `
${esc(status)}${esc(r.studentName)} · ${esc(r.rule)}
${esc(r.roomCode)} ${esc(r.timestamp)} ${esc(r.reporterRoom||'Gestão')}${esc(r.reporterEmail||'')}
${obs}${decision}
${actions}
` }).join(''); refreshPendingCounters(); } function handleApproveReport(id){ if(!appData.isEEB)return; const report=(appData.reviewQueueList||[]).find(r=>String(r.id)===String(id)); if(!report)return; if(!confirm(`Aprovar esta denúncia?\n\n${report.studentName} — ${report.rule}\n${report.roomCode}\n\nApós a aprovação, ela entrará no ranking.`))return; google.script.run .withSuccessHandler(res=>{ if(!res?.success)return showToast('Não foi possível aprovar',res?.message||'A denúncia não foi alterada.','warning'); replaceReportLocally(res.report); if(res.occurrence&&!appData.historyList.some(i=>String(i.id)===String(res.occurrence.id)))appData.historyList.unshift(res.occurrence); refreshPendingCounters();renderReviewQueue();renderMyReports();renderRanking();renderHistory(); showToast('Denúncia aprovada',res.message||'A ocorrência entrou no ranking.','success'); }) .withFailureHandler(e=>showToast('Erro ao aprovar',e.message||String(e),'error')) .aprovarDenunciaPendente(id,''); } function handleDenyReport(id){ if(!appData.isEEB)return; const report=(appData.reviewQueueList||[]).find(r=>String(r.id)===String(id));if(!report)return; const reason=prompt(`Motivo da negativa:\n${report.studentName} — ${report.rule}\n\nO registro ficará arquivado e não afetará o ranking.`,'Não confirmada após análise pedagógica.'); if(reason===null)return; google.script.run .withSuccessHandler(res=>{ if(!res?.success)return showToast('Não foi possível negar',res?.message||'A denúncia não foi alterada.','warning'); replaceReportLocally(res.report);refreshPendingCounters();renderReviewQueue();renderMyReports(); showToast('Denúncia negada',res.message||'Nenhum ponto foi alterado.','warning'); }) .withFailureHandler(e=>showToast('Erro ao negar',e.message||String(e),'error')) .negarDenunciaPendente(id,String(reason||'').trim()); } function handleFalseReport(id){ if(!appData.isEEB)return; const report=(appData.reviewQueueList||[]).find(r=>String(r.id)===String(id));if(!report)return; const reason=prompt(`Justificativa para DENÚNCIA FALSA:\n${report.studentName} — ${report.rule}\n\nEssa decisão pode penalizar a turma do denunciante em 1 ponto.`,''); if(reason===null)return;if(!String(reason).trim())return showToast('Justificativa obrigatória','Informe por que a denúncia foi considerada falsa.','warning'); google.script.run .withSuccessHandler(res=>{ if(!res?.success)return showToast('Não foi possível classificar',res?.message||'A denúncia não foi alterada.','warning'); replaceReportLocally(res.report);if(res.falseReport)appData.falseReportsList.unshift(res.falseReport); refreshPendingCounters();renderReviewQueue();renderMyReports();renderFalseReports();renderRanking(); showToast('Denúncia marcada como falsa',res.message||'A decisão foi registrada.','warning'); }) .withFailureHandler(e=>showToast('Erro na revisão',e.message||String(e),'error')) .marcarDenunciaFalsaPendente(id,String(reason).trim()); }
